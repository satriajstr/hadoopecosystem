01 · HDFS

Simpan data dan buat salinan cadangan

Contoh: catatan imunisasi anak

Sebelum menghitung, kita perlu menyimpan seluruh catatan. HDFS membagi file menjadi potongan yang disebut blok, lalu menyimpan salinannya di beberapa tempat.

Tujuan demo: menunjukkan bahwa file tetap dapat dibaca ketika satu tempat penyimpanan gagal.

Detail teknis: lokasi blok dan kode

Blok adalah potongan byte, sehingga satu baris teks bisa melintasi dua blok. Penggabungan kembali mempertahankan file utuh. Jejak hasil ada pada hdfs/results.json dalam folder sesi.

Kode pembagian blok dan pemulihan
"""Small byte-block model: metadata, replicas, checksums, and node recovery."""
import copy
import hashlib
from pathlib import Path
from prepare_dummy_dataset import save_json

BLOCK_SIZE = 16 * 1024  # Teaching size so this small registry has several blocks.
NODES = ('DN1', 'DN2', 'DN3')


def checksum(data):
    return hashlib.sha256(data).hexdigest()


def write_blocks(source, names, root, block_size=BLOCK_SIZE, replication=2):
    if block_size <= 0 or not 1 <= replication <= len(NODES):
        raise ValueError('Invalid block size or replication factor')
    root = Path(root)
    root.mkdir(parents=True, exist_ok=True)
    manifest = {'block_size_bytes': block_size, 'replication': replication,
                'nodes': {node: 'ONLINE' for node in NODES}, 'files': {}}
    ordinal = 0
    for name in names:
        data = (Path(source)/name).read_bytes()
        blocks = []
        for offset in range(0, len(data), block_size):
            part = data[offset:offset+block_size]
            block_id = f'blk_{ordinal:05d}'
            replicas = [NODES[(ordinal+i) % len(NODES)] for i in range(replication)]
            for node in replicas:
                folder = root/'datanodes'/node
                folder.mkdir(parents=True, exist_ok=True)
                (folder/block_id).write_bytes(part)
            blocks.append({'id': block_id, 'bytes': len(part),
                           'sha256': checksum(part), 'replicas': replicas})
            ordinal += 1
        manifest['files'][name] = {'bytes': len(data), 'sha256': checksum(data), 'blocks': blocks}
    save_json(root/'namenode.json', manifest)
    return manifest


def read_block(root, manifest, block):
    for node in block['replicas']:
        path = Path(root)/'datanodes'/node/block['id']
        if manifest['nodes'][node] != 'ONLINE' or not path.is_file():
            continue
        data = path.read_bytes()
        if len(data) == block['bytes'] and checksum(data) == block['sha256']:
            return data, node
    raise ValueError('No healthy replica for '+block['id'])


def read_file(root, manifest, name):
    # The client gets locations from metadata, then reads bytes from DataNodes.
    chunks, trace = [], []
    for block in manifest['files'][name]['blocks']:
        data, node = read_block(root, manifest, block)
        chunks.append(data)
        trace.append({'block': block['id'], 'node': node})
    result = b''.join(chunks)
    expected = manifest['files'][name]
    if len(result) != expected['bytes'] or checksum(result) != expected['sha256']:
        raise ValueError('Reconstructed file differs from metadata: '+name)
    return result, trace


def repair_replicas(root, manifest):
    repairs = []
    for info in manifest['files'].values():
        for block in info['blocks']:
            data, source = read_block(root, manifest, block)
            healthy = []
            for node in block['replicas']:
                path = Path(root)/'datanodes'/node/block['id']
                if manifest['nodes'][node] == 'ONLINE' and path.is_file() and checksum(path.read_bytes()) == block['sha256']:
                    healthy.append(node)
            for target in NODES:
                if len(healthy) >= manifest['replication']:
                    break
                if target in healthy or manifest['nodes'][target] != 'ONLINE':
                    continue
                folder = Path(root)/'datanodes'/target
                folder.mkdir(parents=True, exist_ok=True)
                (folder/block['id']).write_bytes(data)
                healthy.append(target)
                repairs.append({'block': block['id'], 'source': source, 'target': target})
            block['replicas'] = healthy
    save_json(Path(root)/'namenode.json', manifest)
    return repairs


def demonstrate_storage(source, names, root, destination):
    root, destination = Path(root), Path(destination)
    destination.mkdir(parents=True, exist_ok=True)
    manifest = write_blocks(source, names, root)
    initial = copy.deepcopy(manifest)
    before = {}
    for name in names:
        data, before[name] = read_file(root, manifest, name)
        (destination/name).write_bytes(data)
    # A failed heartbeat excludes DN1; its old files are not usable replicas.
    manifest['nodes']['DN1'] = 'OFFLINE'
    after = {}
    for name in names:
        data, after[name] = read_file(root, manifest, name)
        if data != (destination/name).read_bytes():
            raise ValueError('Read after node failure differs from the initial read')
    repairs = repair_replicas(root, manifest)
    for name in names:
        read_file(root, manifest, name)
    evidence = {'initial': initial, 'after_failure': copy.deepcopy(manifest),
                'read_before_failure': before, 'read_after_failure': after,
                'failed_node': 'DN1', 'repairs': repairs,
                'all_files_match_after_failure': True,
                'replication_restored': all(len(b['replicas']) == manifest['replication']
                    for f in manifest['files'].values() for b in f['blocks'])}
    save_json(root/'results.json', evidence)
    return evidence


def verify_storage(root, destination):
    import json
    root = Path(root)
    manifest = json.loads((root/'namenode.json').read_text(encoding='utf-8'))
    for name in manifest['files']:
        data, _ = read_file(root, manifest, name)
        if data != (Path(destination)/name).read_bytes():
            raise ValueError('Storage snapshot differs from block replicas: '+name)


def print_storage(evidence,detail=False):
    initial = evidence['initial']
    if not detail:
        blocks = sum(len(info['blocks']) for info in initial['files'].values())
        print('Blok:',blocks,'| ukuran:',initial['block_size_bytes']//1024,'KiB | replika:',initial['replication'])
        print('DN1 OFFLINE | file terbaca dari DN2/DN3 | hash cocok')
        print('Replika pengganti:',len(evidence['repairs']),'blok | pulih:',
              'ya' if evidence['replication_restored'] else 'belum')
        return
    print('Blok = potongan file. Replika = salinan potongan itu.')
    print('NameNode mencatat lokasi; DN1, DN2, DN3 adalah tempat penyimpanan (DataNode).')
    print('Ukuran potongan:', initial['block_size_bytes']//1024, 'KiB | salinan per potongan:', initial['replication'])
    for name, info in initial['files'].items():
        print(name, '->', len(info['blocks']), 'blok |', info['bytes'], 'byte')
    for block in initial['files']['children.tsv']['blocks'][:3]:
        print(' ', block['id'], '->', ', '.join(block['replicas']))
    print('DN1 OFFLINE = satu tempat penyimpanan dibuat gagal dalam latihan.')
    print('File tetap terbaca dari salinan di DN2/DN3. Pemeriksaan isi (hash): cocok.')
    print('Salinan pengganti dibuat untuk', len(evidence['repairs']), 'blok.')
    print('Setiap blok kembali memiliki dua salinan:', 'ya' if evidence['replication_restored'] else 'belum')

02 · MAPREDUCE + YARN

Bagi pekerjaan, lalu gabungkan hitungannya

Contoh: catatan imunisasi anak

Data dari tahap sebelumnya berisi 80.137 anak. Sekarang kita menghitung anak yang masuk kelompok analisis, penerima D1, dan penerima D3 untuk setiap kabupaten.

MapReduce membagi dan merangkum pekerjaan. YARN mengatur kapasitas CPU/RAM agar pekerjaan mendapat tempat untuk berjalan.

Lalu YARN melakukan apa?

Bayangkan ada beberapa komputer yang bisa mengerjakan tugas. YARN memeriksa kapasitasnya dan memberi jatah CPU/RAM untuk tiap tugas. Jatah tersebut disebut container.

Nama dalam YARNArti sederhana
ResourceManagerPengatur kapasitas seluruh komputer: memilih di mana tugas mendapat jatah CPU/RAM
ApplicationMasterKoordinator satu pekerjaan: meminta jatah dan memantau tugasnya
NodeManager (NM1, NM2, NM3)Pengelola pada tiap komputer: menjalankan tugas sesuai jatah yang diberikan

Pada contoh ini, NM1 menampung koordinator. NM2 menjalankan Mapper A dan NM3 menjalankan Mapper B. Setelah hasil kedua mapper terkumpul, reducer menghitung totalnya. Jatah dilepas setelah tugas selesai.

Pekerjaan dikirim
Kapasitas disediakan
Tugas berjalan
Selesai, kapasitas dilepas

Kode mapper: baca profil dan keluarkan angka
"""Dummy child TSV -> district key and eligible/D1/D3 counts."""
import sys


def main():
    for line in sys.stdin:
        if not line.strip():
            continue
        fields = line.rstrip('\r\n').split('\t')
        if len(fields)!=10:
            raise ValueError('Registry TSV requires ten columns')
        _,province,district,_,year,due,d1,d2,d3,_ = fields
        year,due,d1,d2,d3 = map(int,(year,due,d1,d2,d3))
        if any(value not in (0,1) for value in (due,d1,d2,d3)) or not d3<=d2<=d1:
            raise ValueError('Invalid eligibility/dose sequence')
        if year==2023 and due==1:
            print(f'{province}|{district}\t1\t{d1}\t{d3}')


if __name__=='__main__':
    main()
Kode reducer: jumlahkan per wilayah
"""Sorted district key, eligible/D1/D3 counts -> coverage and dropout."""
import sys
from decimal import Decimal, ROUND_HALF_UP


def percentage(numerator,denominator):
    return str((Decimal(100)*numerator/denominator).quantize(Decimal('0.01'),rounding=ROUND_HALF_UP)) if denominator else '\\N'


def emit(key,eligible,d1,d3):
    print(f'{key}\t{eligible}\t{d1}\t{d3}\t{percentage(d3,eligible)}\t{percentage(d1-d3,d1)}')


def main():
    current = None
    eligible = d1 = d3 = 0
    for line in sys.stdin:
        if not line.strip():
            continue
        key,n,a,b = line.rstrip('\r\n').split('\t')
        n,a,b = map(int,(n,a,b))
        if n<=0 or not 0<=b<=a<=n:
            raise ValueError('Invalid partial district counts')
        if current is not None and key<current:
            raise ValueError('Reducer input must be sorted by key')
        if current is not None and key!=current:
            emit(current,eligible,d1,d3)
            eligible = d1 = d3 = 0
        current = key
        eligible += n
        d1 += a
        d3 += b
    if current is not None:
        emit(current,eligible,d1,d3)


if __name__=='__main__':
    main()
Kode alokasi kapasitas YARN
"""Resource-budget model tied to actual Python task execution."""
from concurrent.futures import ThreadPoolExecutor, wait, FIRST_COMPLETED
from datetime import datetime, timezone
from time import perf_counter
from uuid import uuid4


class YarnApplication:
    def __init__(self, node_count=3, cpu=1, memory_mb=1024):
        if node_count < 2 or cpu < 1 or memory_mb < 512:
            raise ValueError('The application needs an AM and task resources')
        self.nodes = {f'NM{i+1}': {'cpu': cpu, 'memory_mb': memory_mb,
                      'used_cpu': 0, 'used_memory_mb': 0} for i in range(node_count)}
        self.application_id = 'demo_app_'+uuid4().hex[:10]
        self.events, self.containers = [], {}
        self.state = 'NEW'
        self.started = perf_counter()
        self.record('ResourceManager', 'SUBMITTED', self.application_id)
        self.state = 'SUBMITTED'
        self.record('ResourceManager', 'ACCEPTED', 'ApplicationMaster requested')
        self.state = 'ACCEPTED'
        self.am = self.allocate('ApplicationMaster', 1, 256)
        if self.am is None:
            raise ValueError('Insufficient resources for ApplicationMaster')
        self.state = 'RUNNING'
        self.record('ApplicationMaster', 'RUNNING', self.application_id)

    def record(self, actor, event, detail):
        self.events.append({'sequence': len(self.events)+1, 'actor': actor,
            'event': event, 'detail': detail, 'elapsed_ms': round((perf_counter()-self.started)*1000, 3),
            'utc': datetime.now(timezone.utc).isoformat(),
            'resources': {node: dict(value) for node, value in self.nodes.items()}})

    def allocate(self, task, cpu=1, memory_mb=512):
        for node, resources in self.nodes.items():
            if resources['used_cpu']+cpu > resources['cpu'] or resources['used_memory_mb']+memory_mb > resources['memory_mb']:
                continue
            resources['used_cpu'] += cpu
            resources['used_memory_mb'] += memory_mb
            container = 'container_'+str(len(self.containers)+1).zfill(3)
            self.containers[container] = {'task': task, 'node': node, 'cpu': cpu,
                                          'memory_mb': memory_mb, 'active': True}
            self.record('ResourceManager', 'ALLOCATED', f'{container}: {task} @ {node}; CPU={cpu}, RAM={memory_mb} MiB')
            self.record(node, 'LAUNCHED', f'{container}: {task}')
            return container
        return None

    def release(self, container, status='COMPLETED'):
        info = self.containers[container]
        if not info['active']:
            return
        resources = self.nodes[info['node']]
        resources['used_cpu'] -= info['cpu']
        resources['used_memory_mb'] -= info['memory_mb']
        info['active'] = False
        self.record(info['node'], status, f'{container}: {info["task"]}; resource released')

    def run_tasks(self, tasks):
        if self.state != 'RUNNING':
            raise ValueError('Application is not running')
        # tasks is [(label, callable)]. Only allocated tasks may be submitted.
        pending = list(enumerate(tasks))
        results, active = {}, {}
        for _, (label, _) in pending:
            self.record('ApplicationMaster', 'REQUESTED', label+'; CPU=1, RAM=512 MiB')
        try:
            with ThreadPoolExecutor(max_workers=len(self.nodes)) as pool:
                while pending or active:
                    while pending:
                        index, (label, function) = pending[0]
                        container = self.allocate(label)
                        if container is None:
                            self.record('ResourceManager', 'WAITING', label+'; waiting for free CPU/RAM')
                            break
                        pending.pop(0)
                        active[pool.submit(function)] = (index, container)
                    if not active:
                        raise ValueError('No available node can satisfy the task request')
                    done, _ = wait(active, return_when=FIRST_COMPLETED)
                    for future in done:
                        index, container = active.pop(future)
                        try:
                            results[index] = future.result()
                        except Exception:
                            self.release(container, 'FAILED')
                            raise
                        self.release(container)
        except Exception:
            self.finish(False)
            raise
        return [results[index] for index in range(len(tasks))]

    def finish(self, success=True):
        if self.state in ('FINISHED', 'FAILED'):
            return
        for container in self.containers:
            self.release(container, 'COMPLETED' if success else 'STOPPED')
        self.state = 'FINISHED' if success else 'FAILED'
        self.record('ResourceManager', self.state, 'SUCCEEDED' if success else 'FAILED')

    def evidence(self):
        return {'application_id': self.application_id, 'state': self.state,
                'final_status': 'SUCCEEDED' if self.state == 'FINISHED' else self.state,
                'resource_units': 'logical CPU and MiB for the teaching scheduler',
                'nodes': self.nodes, 'containers': self.containers, 'events': self.events}


def print_yarn(evidence,detail=False):
    if not detail:
        print('YARN:',evidence['final_status'])
        print('Tugas:',', '.join(container['task']+' -> '+container['node']
              for container in evidence['containers'].values() if container['task']!='ApplicationMaster'))
        if any(event['event']=='WAITING' for event in evidence['events']):
            print('WAITING: Mapper B menunggu Mapper A selesai.')
        free = all(node['used_cpu']==0 and node['used_memory_mb']==0 for node in evidence['nodes'].values())
        print('Resource dilepas:', 'ya' if free else 'belum')
        return
    print('YARN mengatur jatah CPU/RAM agar setiap tugas mendapat tempat untuk berjalan.')
    print('1. Pekerjaan dikirim dan diterima. ID:', evidence['application_id'])
    print('2. Koordinator (ApplicationMaster) mendapat tempat di NM1.')
    print('3. Pengatur kapasitas (ResourceManager) memberi jatah; NodeManager menjalankan tugas:')
    for container in evidence['containers'].values():
        if container['task'] != 'ApplicationMaster':
            print('  ',container['task'],'->',container['node'],
                  '| jatah:',container['cpu'],'CPU dan',container['memory_mb'],'MiB RAM')
    if any(event['event']=='WAITING' for event in evidence['events']):
        print('   WAITING = menunggu: Mapper B baru mendapat tempat setelah Mapper A selesai.')
    if evidence['final_status']=='SUCCEEDED':
        print('4. Semua tugas selesai. FINISHED / SUCCEEDED = pekerjaan berhasil.')
    else:
        print('4. Status pekerjaan:',evidence['final_status'])
    free = all(node['used_cpu']==0 and node['used_memory_mb']==0 for node in evidence['nodes'].values())
    print('Jatah CPU/RAM sudah dilepas:', 'ya' if free else 'belum')
    if detail:

Detail teknis: query SQL dan tabel Hive

SELECT memilih hasil; WHERE membatasi anak yang dihitung; GROUP BY mengelompokkan wilayah; SUM menjumlahkan status dosis. NULLIF menjaga agar query tidak membagi angka dengan nol.

Query yang menghitung cakupan dan dropout
-- Synthetic child registry, one row per child. Same query tested in SQLite.
-- Eligibility is an invented teaching flag, not a clinical schedule.
WITH cohort AS (
    SELECT * FROM vaccinations
    WHERE birth_year = 2023 AND dose3_due = 1
)
SELECT province, district, COUNT(*) AS eligible_children,
       SUM(dose1) AS dose1_children, SUM(dose3) AS dose3_children,
       ROUND(100.0 * SUM(dose3) / COUNT(*), 2) AS coverage_pct,
       ROUND(100.0 * (SUM(dose1) - SUM(dose3))
             / NULLIF(SUM(dose1), 0), 2) AS dropout_pct,
       SUM(CASE WHEN dose3 = 0 THEN 1 ELSE 0 END) AS missed_dose3
FROM cohort
GROUP BY province, district
ORDER BY coverage_pct ASC, province ASC, district ASC;
Query putus lanjut keseluruhan

       COALESCE(SUM(dose1), 0) AS dose1_children,
       COALESCE(SUM(dose3), 0) AS dose3_children,

Dalam Pig Latin: LOAD berarti baca file, FILTER berarti pilih baris sesuai kondisi, dan STORE berarti simpan hasil. Data tahun lahir lain tetap ada di sumber, tetapi tidak masuk daftar yang dipilih.

Kode Pig Latin: baca → pilih → simpan
-- Input children.tsv has ten columns and no header.
raw = LOAD '$INPUT' USING PigStorage('\t') AS (
    child_id:chararray, province:chararray, district:chararray, birth_date:chararray,
    birth_year:int, dose3_due:int, dose1:int, dose2:int, dose3:int, last_updated:chararray
);
cohort = FILTER raw BY birth_year == 2023 AND dose3_due == 1;
missed = FILTER cohort BY dose3 == 0;
dropout = FILTER missed BY dose1 == 1;
not_started = FILTER missed BY dose1 == 0;
STORE cohort INTO '$OUTPUT/cohort' USING PigStorage('\t');
STORE missed INTO '$OUTPUT/missed-dose3' USING PigStorage('\t');
STORE dropout INTO '$OUTPUT/dropout-d1-d3' USING PigStorage('\t');
STORE not_started INTO '$OUTPUT/not-started-d1' USING PigStorage('\t');

HBase disebut wide-column karena data diatur menurut row key dan kelompok kolom. Tanggal vaksinasi di value berbeda dari timestamp versi sel. Kode PUT menyimpan catatan; GET mengambilnya.

Coba ID lain: catatan kosong atau anak tidak ditemukan

D000046 adalah anak yang terdaftar tetapi belum mempunyai catatan dosis. TIDAK_ADA adalah ID yang tidak ada dalam data. Keduanya harus dibedakan.

Untuk melihat anak yang terdaftar dengan riwayat kosong, ketik:

python code/demo.py hbase --child D000046

Untuk melihat hasil ID yang tidak ditemukan, ketik:

python code/demo.py hbase --child TIDAK_ADA

PENUTUP

Apa yang sudah kita pelajari?

Kita mengikuti catatan imunisasi dari penyimpanan sampai daftar anak dan riwayat dosisnya. Setiap alat membantu menjawab satu bagian dari pertanyaan awal.

AlatIngat tugas utamanya
HDFSSimpan potongan file dan salinannya agar data tetap dapat dibaca
YARNAtur jatah CPU/RAM untuk menjalankan pekerjaan
MapReduceBagi data, kelompokkan, lalu jumlahkan hasilnya
HiveGunakan SQL untuk meringkas dan menganalisis data
PigPilih data langkah demi langkah dan simpan daftar hasilnya
HBaseCari profil dan riwayat berdasarkan satu ID anak
Kalimat untuk presentasi: “Dari 80.137 anak, kita memilih 20.035 yang memenuhi syarat. Sebanyak 10.655 sudah menerima dosis ketiga dan 9.380 belum. Ringkasan kabupaten membantu memilih wilayah yang perlu diperhatikan, sedangkan daftar ID membantu menelusuri riwayat tiap anak.”

Hubungannya dengan 5V Big Data

5V membantu menjelaskan tantangan dan manfaat data. Sampel kecil ini digunakan untuk mengikuti alur; skala besar dan pembaruan yang terus masuk menjadi konteks pengembangannya.

5VArti sederhanaPada contoh imunisasi
VolumeBanyaknya dataRegistri nasional dapat berisi banyak anak; latihan ini memakai 80.137
VelocitySeberapa cepat data masuk/diperbaruiCatatan baru dapat datang setiap hari; latihan ini memakai satu kumpulan data
VarietyBeragam bentuk dan sumber dataProfil anak dan catatan dosis memiliki kolom berbeda; ganti format file saja tidak menambah keragaman sumber
VeracitySeberapa dapat dipercaya datanyaPeriksa ID unik, urutan tanggal/dosis, dan kecocokan profil dengan riwayat
ValueManfaat yang diperolehTemukan wilayah bercakupan rendah dan anak yang belum menerima D3

Hasil ini membantu menjelaskan tindak lanjut imunisasi. Prediksi wabah merupakan pekerjaan lanjutan yang memerlukan data tambahan, model, dan pengujian.

Ingin mengulang latihan?

Untuk melihat seluruh tahap sekaligus, jalankan perintah di bawah. all berarti semua tahap. Program membuat sesi baru dan menyimpan hasil setiap percobaan dalam folder baru. Saat presentasi, jalankan perintah per tahap agar mudah dijelaskan.

python code/demo.py all

Nama folder hasil muncul di akhir proses. Memulai lagi dengan storage membuat sesi baru. Mengulang tahap lain menggunakan sesi yang sedang aktif tanpa menimpa hasil sebelumnya.
