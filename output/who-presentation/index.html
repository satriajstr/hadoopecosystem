Peran YARN pada Hadoop

YARN mengalokasikan CPU/RAM untuk ApplicationMaster dan container tugas. Framework MapReduce menangani pembagian input, shuffle/sort, dan reducer. Alur pemrosesan meliputi dua mapper, pengelompokan key, dan reducer. YARN menyediakan resource yang dibutuhkan job.

Kode mapper imunisasi
"""Streaming mapper: five-country annual DTP3 estimates, emit sum/count."""
import sys
from decimal import Decimal

COUNTRIES = {'IDN': 'Indonesia', 'MYS': 'Malaysia', 'THA': 'Thailand',
             'VNM': 'Viet Nam', 'PHL': 'Philippines'}
YEARS = tuple(range(2020, 2026))


def map_record(line):
    fields = line.rstrip('\r\n').split('\t')
    if len(fields) != 10:
        raise ValueError('Expected ten columns in the WUENIC TSV')
    area, _, indicator, sex, year, coverage, unit, status, revision, age = fields
    if (area not in COUNTRIES or indicator != 'IM_DTP3' or sex != '_T'
            or unit != 'PCNT' or status != 'E' or int(year) not in YEARS):
        return None
    if coverage in ('', '\\N'):
        return None
    value = Decimal(coverage)
    if not value.is_finite() or not 0 <= value <= 100:
        raise ValueError('Coverage must be a finite percentage between 0 and 100')
    # Include metadata in the key, so different source revisions/ages never mix.
    key = '|'.join((area, indicator, sex, unit, status, revision, age))
    return f'{key}\t{value}\t1\n'


def main():
    for line in sys.stdin:
        if line.strip():
            mapped = map_record(line)
            if mapped is not None:
                sys.stdout.write(mapped)


if __name__ == '__main__':
    main()
Kode reducer imunisasi
"""Sorted streaming input: key TAB sum_of_year_percentages TAB year_count."""
import sys
from decimal import Decimal, ROUND_HALF_UP


def emit(key, total, count):
    mean = (total / count).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
    print(f'{key}\t{total}\t{count}\t{mean}')


current_key = None
total = Decimal(0)
count = 0
for line in sys.stdin:
    if not line.strip():
        continue
    key, value_text, count_text = line.rstrip('\r\n').split('\t')
    value, n = Decimal(value_text), int(count_text)
    if not value.is_finite() or n <= 0:
        raise ValueError('Reducer needs a finite sum and a positive count')
    if current_key is not None and key != current_key:
        emit(current_key, total, count)
        total, count = Decimal(0), 0
    current_key = key
    total += value
    count += n
if current_key is not None:
    emit(current_key, total, count)

SELECT area_code, area_name, source_revision, age_group,
       SUM(coverage_pct) AS sum_of_year_percentages,
       COUNT(*) AS year_count, ROUND(AVG(coverage_pct),2) AS mean_yearly_dtp3_pct
FROM real_coverage_long
WHERE area_code IN ('IDN','MYS','THA','VNM','PHL')
      AND indicator = 'IM_DTP3' AND sex = '_T' AND unit = 'PCNT'

Interpretasi Indonesia 2025

(85 - 82) / 85 * 100 = 3,53%

DTP3 lebih rendah tiga poin persentase daripada DTP1. Proxy ini memakai dua estimasi agregat; tidak melacak anak yang sama. Mean DTP3 2020–2025 tetap 79,67%, berbeda dari cakupan 2025 sebesar 82%.

Mean dan jumlah tahun untuk kelima negara cocok dengan hasil MapReduce lokal. Perintah manual memeriksa kesamaan ini apabila MapReduce sudah dijalankan dalam sesi tersebut.

Lima query lengkap
-- Query 1: jumlah observasi, bukan jumlah anak.
SELECT COUNT(*) AS observation_rows
FROM real_coverage_long;

-- Query 2: lima negara pada tahun yang sama; agregat regional tidak ikut.
SELECT area_code, area_name, year, indicator, coverage_pct
FROM real_coverage_long
WHERE area_code IN ('IDN','MYS','THA','VNM','PHL') AND year = 2025 AND sex = '_T'
      AND unit = 'PCNT' AND observation_status = 'E'
ORDER BY area_code, indicator;

-- Query 3: mean per negara. Keenam tahun berbobot sama, bukan mean global.
SELECT area_code, area_name, source_revision, age_group,
       SUM(coverage_pct) AS sum_of_year_percentages,
       COUNT(*) AS year_count, ROUND(AVG(coverage_pct),2) AS mean_yearly_dtp3_pct
FROM real_coverage_long
WHERE area_code IN ('IDN','MYS','THA','VNM','PHL')
      AND indicator = 'IM_DTP3' AND sex = '_T' AND unit = 'PCNT'
      AND observation_status = 'E' AND year BETWEEN 2020 AND 2025
GROUP BY area_code, area_name, source_revision, age_group
ORDER BY area_code;

-- Query 4: tren per negara; jangan menjumlahkan persentase sebagai cakupan.
SELECT area_code, area_name, year, coverage_pct AS dtp3_pct
FROM real_coverage_long
WHERE area_code IN ('IDN','MYS','THA','VNM','PHL') AND indicator = 'IM_DTP3' AND sex = '_T'
      AND unit = 'PCNT' AND observation_status = 'E'
      AND year BETWEEN 2020 AND 2025
ORDER BY year, area_code;

-- Query 5: pasangan indikator dan proxy dropout agregat, bukan riwayat anak.
WITH paired AS (
    SELECT area_code, area_name, year, source_revision, age_group,
           MAX(CASE WHEN indicator = 'IM_DTP1' THEN coverage_pct END) AS dtp1_pct,
           MAX(CASE WHEN indicator = 'IM_DTP3' THEN coverage_pct END) AS dtp3_pct
    FROM real_coverage_long
    WHERE area_code IN ('IDN','MYS','THA','VNM','PHL') AND sex = '_T' AND unit = 'PCNT'
          AND observation_status = 'E'
          AND year BETWEEN 2020 AND 2025
          AND indicator IN ('IM_DTP1', 'IM_DTP3')
    GROUP BY area_code, area_name, year, source_revision, age_group
)
SELECT area_code, area_name, year, dtp1_pct, dtp3_pct,
       ROUND(dtp1_pct - dtp3_pct, 2) AS gap_percentage_points,
       CASE WHEN dtp1_pct > 0 AND dtp3_pct <= dtp1_pct
            THEN ROUND(100.0 * (dtp1_pct - dtp3_pct) / NULLIF(dtp1_pct, 0), 2)
            ELSE NULL END AS dropout_proxy_pct
FROM paired
WHERE dtp1_pct IS NOT NULL AND dtp3_pct IS NOT NULL
ORDER BY year, area_code;

External table memberi struktur pada file, lalu SQL memilih dan meringkas data sesuai pertanyaan analisis.

Kode Pig Latin yang disiapkan
-- Input: tabel WUENIC sebelum filter, berupa TSV tanpa header.
raw = LOAD '$INPUT' USING PigStorage('\t') AS (
    area_code:chararray, area_name:chararray, indicator:chararray, sex:chararray,
    year:int, coverage_pct:double, unit:chararray, observation_status:chararray,
    source_revision:chararray, age_group:chararray
);
selected = FILTER raw BY (area_code == 'IDN' OR area_code == 'MYS'
    OR area_code == 'THA' OR area_code == 'VNM' OR area_code == 'PHL') AND sex == '_T'
    AND unit == 'PCNT' AND observation_status == 'E'
    AND year >= 2020 AND year <= 2025
    AND (indicator == 'IM_DTP1' OR indicator == 'IM_DTP3')
    AND coverage_pct IS NOT NULL AND coverage_pct >= 0 AND coverage_pct <= 100;
by_country_year = GROUP selected BY (area_code, year);
counts = FOREACH by_country_year GENERATE FLATTEN(group) AS (area_code, year), COUNT_STAR(selected) AS indicator_rows;
ordered = ORDER counts BY area_code ASC, year ASC;
STORE selected INTO '$OUTPUT/filtered' USING PigStorage('\t');
STORE ordered INTO '$OUTPUT/by-country-year' USING PigStorage('\t');

Pig Latin mengekspresikan urutan LOAD, FILTER, GROUP, dan STORE.

Kode reducer Word Count
#!/usr/bin/env python3
"""Input harus sudah dikelompokkan/diurutkan berdasarkan key oleh shuffle."""
import sys

current_word = None
total = 0
for line in sys.stdin:
    if not line.strip():
        continue
    word, count = line.rstrip("\n").split("\t", 1)
    if current_word is not None and word != current_word:
        print(f"{current_word}\t{total}")
        total = 0
    current_word = word
    total += int(count)

if current_word is not None:
    print(f"{current_word}\t{total}")
